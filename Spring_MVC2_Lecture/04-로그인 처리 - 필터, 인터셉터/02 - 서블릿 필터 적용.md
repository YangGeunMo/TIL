# 서블릿 필터 - 요청 로그 


## 로그 필터 개발

~~~java
@Slf4j
//필터를 사용하려면 인터페이스 구현
public class LogFilter implements Filter {
    @Override
    // 필터 객체가 생성되어 초기화될 때 한 번 호출된다.
    public void init(FilterConfig filterConfig) throws ServletException {
        log.info("log filter init");
    }

    @Override

    //HTTP 요청이 들어오면 doFilter가 호출됨.
    // 요청이 들어올 때마다 호출된다. ServletRequest로 받는 이유는 HTTP가 아닌 요청도 고려했기 때문이다.
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain) throws IOException, ServletException {
        log.info("log filter doFilter");
        //http를 사용하려면 HttpServletRequest로 다운 캐스팅 한다.
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        String requestURI = httpRequest.getRequestURI();
        //http 요청을 구분하기 위해 임의 값 지정
        String uuid = UUID.randomUUID().toString();

        try {
            log.info("REQUEST [{}] [{}]", uuid, requestURI);
            //필터가 있으면 필터를 호출, 없으면 서블릿을 호출한다. 없을시 다음 단계로 진행이 안된다.
            chain.doFilter(request,response);
        } catch (Exception e) {

        }finally {
            log.info("RESPONSE [{}] [{}]", uuid, requestURI);
        }
    }

    @Override
    //서블릿 컨테이너 종료될때 호출
    public void destroy() {
        log.info("log filter destroy");
    }
}
~~~

### doFilter 흐름

- doFilter는 HTTP 요청이 들어올떄마다 메서드가 호출된다.

- 요청이 들어올때마다 REQUEST 로그를 남긴다.
  
- chain.doFilter(request,response);는 다음 필터 또는 DespatcherServlet에 전달한다.

- chain.doFilter가 없으면 다음 컨트롤러로 전달할 수 없다. 


## 필터 등록

- 스프링 부트를 사용해서 `FilterRegistrationBean` 를 사용해 등록하면 된다.
  - `setFilter(new LogFilter())` : 등록할 필터를 지정한다.
  - `setOrder(1)` : 필터는 체인으로 동작한다. 따라서 순서가 필요하다. 낮을 수록 먼저 동작한다.
  - `addUrlPatterns("/*")` : 필터를 적용할 URL 패턴을 지정한다. 한번에 여러 패턴을 지정할 수 있다.

~~~java
// 스프링 설정 클래스임을 나타낸다. 이 클래스 안의 @Bean 메서드가 반환하는 객체를 스프링 빈으로 등록한다.
@Configuration
public class WebConfig{

    @Bean
    // 필터를 서블릿 컨테이너에 등록하는 설정이다.
    public FilterRegistrationBean logFilter() {
        //필터 등록
        FilterRegistrationBean<Filter> filterRegistrationBean = new FilterRegistrationBean<>();
        //등록할 필터 지정
        filterRegistrationBean.setFilter(new LogFilter());
        //필터는 체인으로 동작하므로 순서가 필요하다. 낮을 수록 먼져 동작
        filterRegistrationBean.setOrder(1);
        //필터를 적용할 url 패턴을 지정한다. 한번에 여러 패턴을 지정할 수 있다.
        filterRegistrationBean.addUrlPatterns("/*");
        return filterRegistrationBean;
    }
}
~~~

### 등록되는 과정
 - 앱이 시작되면 @Configuration 클래스의 @Bean 메서드가 호출된다.
 
 -   반환된 FilterRegistrationBean을 스프링이 빈으로 보관
   
 -   스프링 부트가 서블릿 컨테이너를 만들 때 이 빈을 찾아 필터 체인에 추가
   
 -   이후 요청이 들어오면 setOrder 순서대로, addUrlPatterns에 맞는 요청에 대해 doFilter가 호출

## 서블릿 필터 - 로그인 인증 체크

~~~java
@Slf4j
public class LoginCheckFilter implements Filter {


    //화이트 리스트에 등록된 url은 로그인없이 접근 가능
    private static final String[] whitelist = {"/", "/members/add", "/login", "/logout", "/css/*"};


    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain) throws IOException, ServletException {

        //http를 사용하기 위해 다운 캐스팅
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        //요청보낸 url 저장
        String requestURI = httpRequest.getRequestURI();
        HttpServletResponse httpResponse = (HttpServletResponse) response;
        
        try{
            log.info("인증 체크 필터 시작{}", requestURI);
            //화이트 리스트에 없는 경로일때 검사
            if (isLoginCheckPath(requestURI)) {
                log.info("인증 체크 로직 실행{}", requestURI);
                //세션이 없으면 null 반환
                HttpSession session = httpRequest.getSession(false);
                if (session == null || session.getAttribute(SessionConst.LOGIN_MEMBER) == null) {

                    log.info("미인증 사용자 요청 {}", requestURI);
                    //미인증 사용자는 로그인 화면으로 리다이렉트한다. 원래 가려던 URL(requestURI)은 redirectURL 파라미터로 함께 보내서, 로그인 후 그 페이지로 돌아갈 수 있게 한다.
                    httpResponse.sendRedirect("/login?redirectURL=" + requestURI);
                    // 여기서 끝내야 아래 chain.doFilter가 호출되지 않아 컨트롤러로 가지 않는다.
                    return;
                }
            }
            // 검사를 통과했거나 검사 대상이 아니면 다음 필터 또는 DispatcherServlet로 요청을 전달한다.
            chain.doFilter(request,response);
        } catch (Exception e) {
            throw e; //예외를 로깅 가능 하지만, 톰켓까지 예외를 보내야 한다.
        }finally {
            log.info("인증 체크 필터 종료{}", requestURI);
        }

    }
    /**
     * 화이트리스트에 포함되지 않은 경로이면 true를 반환한다. (true면 로그인 검사 대상)
     * 화이트리스트에 있는 경로는 false를 반환해서 인증 체크를 하지 않는다.
     */
    private boolean isLoginCheckPath(String requestURI) {
        return !PatternMatchUtils.simpleMatch(whitelist, requestURI);
    }
}
~~~

- 인증 필터를 적용해도 홈, 회원가입, 로그인 화면에는 접근이 가능해야 한다. 그래서 화이트 리스트에 있는 경로는 인증과 상관없이 항상 허용한다.

- 미인증 사용자는 로그인 화면으로 리다이렉트한다. 그런데 로그인 이후 다시 홈으로 이동하면 원하는 경로를 다시 찾아가야 하는 불편함이 있다.

- 예시) 상품 관리 화면을 들어갔는데 로그인 화면으로 이동하면 로그인 후 상품 관리 화면으로 자동으로 이동하게 해야 한다.

- 이런 기능을 `requestURI` 를 `/login` 에 쿼리 파라미터로 함께 전달한다.

## 로그인 인증 체크 등록 
~~~java
    @Bean
    // 필터를 서블릿 컨테이너에 등록하는 설정이다.
    public FilterRegistrationBean logCheckFilter() {
~~~

## RedirectURL 처리
~~~java
@PostMapping("/login")
    public String loginV4(@Valid @ModelAttribute LoginForm form, BindingResult bindingResult,
            //쿼리 파라미터를 redirectURL변수에 저장
            @RequestParam(defaultValue = "/") String redirectURL, HttpServletRequest request) {
        if (bindingResult.hasErrors()) {
            return "login/loginForm";
        }

        Member loginMember = loginService.login(form.getLoginId(), form.getPassword());

        if (loginMember == null) {
            bindingResult.reject("loginFail", "아이디 또는 비밀번호가 맞지 않습니다.");
            return "login/loginForm";
        }

        //로그인 성공 처리
        //세션이 있으면 있는 세션 반환, 없으면 신규 세션 생성
        HttpSession session = request.getSession();
        //세션에 로그인 회원 정보를 보관
        session.setAttribute(SessionConst.LOGIN_MEMBER, loginMember);
        
        //로그인 성공 후 사용자가 처음 가려던 페이지로 이동
        return "redirect:" + redirectURL;
    }
~~~

- 미인증 사용자가 /items에 접근하면 로그인 체크 필터가 /login?redirectURL=/items로 리다이렉트한다.

- 로그인 화면(GET /login)은 redirectURL 값을 hidden 필드에 담아 폼을 보낸다.

- 로그인 처리(POST /login)는 @RequestParam으로 redirectURL을 받아, 로그인 성공 시 redirect: + 그 경로로 이동한다.

## 전체 흐름 정리
예시) /items 요청
- 전체 흐름을 보면 http 요청으로 /items를 서버에 요청했는데  was를 걸쳐 필터에서 doFilter메서드가 실행되고 인증 체크 필터를 시작해서 세션이 없으면 /login?redirectURL=" + requestURI로 이동하고 종료한다.

- 만약 로그인 할경우 @RequestParam(defaultValue = "/") String redirectURL로 쿼리 파라미터를 redirectURL변수에 저장하고 로그인 성공 후 사용자가 처음 가려던 페이지로 이동한다
